<script>
  // Old addresses listed in _data/redirects.yml. Netlify redirects them on the server (/_redirects) and this page is never
  // shown for them; on hosts without server-side redirects (GitHub Pages) a missing address lands here. The check runs
  // before the rest of the page is read: when the address has moved, the "Redirecting" notice is shown instead of the
  // "not found" content (CSS: html.ivl-redirecting) and the browser goes to the new location.
  (function () {
    const redirects = {{ site.data.redirects | jsonify }};
    const path = decodeURIComponent(window.location.pathname);
    const search = new URLSearchParams(window.location.search);
    const base = {{ site.baseurl | jsonify }};
    const strip = (p) => p.replace(/\/+$/, "");
    for (const r of redirects) {
      const [from, query] = r.from.split("?");
      if (strip(base + from) !== strip(path)) continue;
      // address parameters: "key=:name" matches any value (reused as ":name" in "to"), "key=value" must be equal
      const values = {};
      const ok = !query || query.split("&").every((pair) => {
        const [key, value] = pair.split("=");
        if (!search.has(key)) return false;
        if (value.startsWith(":")) return (values[value] = search.get(key)), true;
        return search.get(key) === value;
      });
      if (!ok) continue;
      let to = r.to.replace(/:([A-Za-z_]+)/g, (m, name) => (":" + name in values ? encodeURIComponent(values[":" + name]) : m));
      to = to.includes("://") ? to : base + to;
      window.ivlRedirectTo = to;
      document.documentElement.classList.add("ivl-redirecting");
      document.title = "Redirecting… | " + {{ site.title | jsonify }};
      window.location.replace(to);
      return;
    }
  })();
</script>
